class Cart(ProHandle):
    def __init__(self,user_id):
        self.user_id = user_id
    def add_cart(self):
        pro_name = input ("Enter Pro Name: ")
        price = input("Enter Pro Price: ")
        with open("Product.txt","r") as file:
            list = file.readlines()
        
        flag = False
        new_pro = None
        with open("Product.txt","w") as file:

            for lst in list:
                item = lst.strip().split(",")
                if item[1] == pro_name and item[2] == price and int(item[-1]) > 0:
                    flag = True
                    new_pro = item
                    q = int(item[-1]) - 1
                    file.write(f"{item[0]},{item[1]},{item[2]},{q}\n")
                else:
                    file.write(lst)
            if flag:
                with open("Cart.txt","r") as file:
                    list = file.readlines()
                flag = False
                with open("Cart.txt","w") as file:
                    for lst in list:
                        item = lst.strip().split(",")
                        if item[2] == pro_name and item[3] == price and self.user_id == item[0]:
                            flag = True
                            q = int(item[-1]) + 1
                            file.write(f"{item[0]},{item[1]},{item[2]},{item[3]},{q}\n")
                        else:
                            file.write(lst)
                if flag == False:
                    new_pro[-1] = "1"
                    new_pro.insert(0,self.user_id)
                    with open("Cart.txt","a") as file:
                        file.write(",".join(new_pro)+"\n")
            else:
                print("Product OutOf Stock!")

    def dec_Cart(self):

                    flag = True
